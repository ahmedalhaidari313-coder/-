/// Simple key/value settings (language, theme, PIN, currency, owner info).
/// Relational data lives in SQLite, never here.
class StorageService extends GetxService {
  late final GetStorage _box;

  Future<StorageService> init() async {
    _box = GetStorage();
    return this;
  }

  // ---- language
  Locale get locale => (_box.read<String>('lang') ?? 'ar') == 'ar'
      ? const Locale('ar', 'SA')
      : const Locale('en', 'US');

  Future<void> setLang(String code) async {
    await _box.write('lang', code);
    Get.updateLocale(locale);
  }

  // ---- theme
  ThemeMode get themeMode =>
      (_box.read<bool>('dark') ?? false) ? ThemeMode.dark : ThemeMode.light;

  Future<void> setDark(bool v) async {
    await _box.write('dark', v);
    Get.changeThemeMode(themeMode);
  }

  // ---- PIN (salted SHA-256 hash, never the raw PIN)
  bool get hasPin => (_box.read<String>('pin_hash') ?? '').isNotEmpty;

  String _hash(String pin) {
    const salt = 'ptms.v1.';
    return sha256.convert(utf8.encode('$salt$pin')).toString();
  }

  Future<void> setPin(String pin) => _box.write('pin_hash', _hash(pin));
  Future<void> removePin() => _box.remove('pin_hash');
  bool verifyPin(String pin) => _box.read<String>('pin_hash') == _hash(pin);

  // ---- currency & owner
  String get currency =>
      _box.read<String>('currency') ?? AppConstants.defaultCurrency;
  Future<void> setCurrency(String v) => _box.write('currency', v);

  Map<String, dynamic> get owner =>
      Map<String, dynamic>.from(_box.read('owner') ?? {});
  Future<void> setOwner(Map<String, dynamic> v) => _box.write('owner', v);
}

class DatabaseService extends GetxService {
  late final Database db;
  late final String dbPath;

  Future<DatabaseService> init() async {
    final dir = await getApplicationDocumentsDirectory();
    dbPath = p.join(dir.path, AppConstants.dbName);
    db = await openDatabase(
      dbPath,
      version: AppConstants.dbVersion,
      onConfigure: (d) => d.execute('PRAGMA foreign_keys = ON'),
      onCreate: _onCreate,
    );
    return this;
  }

  /// Copies the database file so the owner can save it outside the app.
  Future<File> createBackupCopy() async {
    final dir = await getTemporaryDirectory();
    final stamp = DateTime.now().toIso8601String().replaceAll(':', '-');
    final target = p.join(dir.path, 'ptms_backup_$stamp.db');
    await db.execute('PRAGMA wal_checkpoint(FULL)');
    return File(dbPath).copy(target);
  }

  Future<void> _onCreate(Database d, int v) async {
    final batch = d.batch();

    batch.execute('''
      CREATE TABLE properties(
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        description TEXT,
        address TEXT,
        floors INTEGER DEFAULT 1,
        notes TEXT,
        is_active INTEGER NOT NULL DEFAULT 1,
        created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
      )''');

    batch.execute('''
      CREATE TABLE units(
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        property_id INTEGER NOT NULL REFERENCES properties(id) ON DELETE RESTRICT,
        number TEXT NOT NULL,
        floor INTEGER,
        type TEXT,
        area REAL,
        base_rent REAL NOT NULL DEFAULT 0,
        status TEXT NOT NULL DEFAULT 'available',
        description TEXT,
        notes TEXT,
        created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
        UNIQUE(property_id, number)
      )''');

    batch.execute('''
      CREATE TABLE tenants(
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        full_name TEXT NOT NULL,
        phone TEXT,
        national_id TEXT,
        address TEXT,
        notes TEXT,
        created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
      )''');

    batch.execute('''
      CREATE TABLE contracts(
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        tenant_id INTEGER NOT NULL REFERENCES tenants(id) ON DELETE RESTRICT,
        unit_id INTEGER NOT NULL REFERENCES units(id) ON DELETE RESTRICT,
        status TEXT NOT NULL DEFAULT 'draft',
        start_date TEXT NOT NULL,
        end_date TEXT NOT NULL,
        rent_amount REAL NOT NULL,
        deposit_amount REAL NOT NULL DEFAULT 0,
        payment_cycle TEXT NOT NULL DEFAULT 'monthly',
        due_day INTEGER NOT NULL DEFAULT 1,
        payment_method TEXT,
        renewal_terms TEXT,
        termination_terms TEXT,
        terms_text TEXT,
        owner_snapshot TEXT,
        tenant_snapshot TEXT,
        pdf_path TEXT,
        created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
      )''');

    batch.execute('''
      CREATE TABLE services(
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        billing_type TEXT NOT NULL,
        split_method TEXT NOT NULL DEFAULT 'direct',
        amount REAL NOT NULL DEFAULT 0,
        property_id INTEGER REFERENCES properties(id) ON DELETE RESTRICT,
        unit_id INTEGER REFERENCES units(id) ON DELETE RESTRICT,
        is_active INTEGER NOT NULL DEFAULT 1,
        notes TEXT,
        created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
      )''');

    batch.execute('''
      CREATE TABLE service_readings(
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        service_id INTEGER NOT NULL REFERENCES services(id) ON DELETE CASCADE,
        unit_id INTEGER NOT NULL REFERENCES units(id) ON DELETE RESTRICT,

    batch.execute('''

        id INTEGER PRIMARY KEY AUTOINCREMENT,
        tenant_id INTEGER NOT NULL REFERENCES tenants(id) ON DELETE RESTRICT,
        contract_id INTEGER NOT NULL REFERENCES contracts(id) ON DELETE RESTRICT,
        obligation_id INTEGER NOT NULL REFERENCES obligations(id) ON DELETE RESTRICT,
        paid_on TEXT NOT NULL,
        amount REAL NOT NULL,
        method TEXT NOT NULL DEFAULT 'cash',
        reference TEXT,
        notes TEXT,
        created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP

    batch.execute('''
      CREATE TABLE documents(
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        owner_type TEXT NOT NULL,

        kind TEXT NOT NULL,
        file_path TEXT NOT NULL,
        mime TEXT,
        size_bytes INTEGER,
        created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP

  }

import 'dart:math';

import 'package:get/get.dart';
import 'package:sqflite/sqflite.dart';

import '../constants/app_constants.dart';
import 'database_service.dart';

/// Creates monthly rent obligations for active contracts.
/// Runs on every app start and catches up on all missed months.
/// Safe to run repeatedly: a UNIQUE index (contract, period, type) plus
/// INSERT OR IGNORE prevents duplicates.
class ObligationService extends GetxService {
  DatabaseService get _db => Get.find<DatabaseService>();

  static String periodOf(DateTime d) =>
      '${d.year}-${d.month.toString().padLeft(2, '0')}';

  static String _iso(DateTime d) =>
      '${d.year}-${d.month.toString().padLeft(2, '0')}-${d.day.toString().padLeft(2, '0')}';

  /// Returns the number of obligations created.
  Future<int> runCatchUp({DateTime? now}) async {
    now ??= DateTime.now();
    final db = _db.db;
    var created = 0;

    await db.transaction((txn) async {
      final contracts = await txn.query('contracts',
          where: 'status = ?', whereArgs: [ContractStatus.active]);

      for (final c in contracts) {
        final start = DateTime.parse(c['start_date'] as String);
        final end = DateTime.parse(c['end_date'] as String);
        final dueDay = (c['due_day'] as int?) ?? 1;
        final unit = (await txn.query('units',
                where: 'id = ?', whereArgs: [c['unit_id']], limit: 1))
            .first;

        var cursor = DateTime(start.year, start.month);
        final last = DateTime(now!.year, now.month);
        final lastForContract = DateTime(end.year, end.month);

        while (!cursor.isAfter(last) && !cursor.isAfter(lastForContract)) {
          final daysInMonth = DateTime(cursor.year, cursor.month + 1, 0).day;
          final due =
              DateTime(cursor.year, cursor.month, min(dueDay, daysInMonth));

          // TODO(v1.1): prorate first/last month if the owner chooses to.
          final id = await txn.insert(
            'obligations',
            {
              'tenant_id': c['tenant_id'],
              'contract_id': c['id'],
              'property_id': unit['property_id'],
              'unit_id': c['unit_id'],
              'type': ObligationType.rent,
              'period': periodOf(cursor),
              'due_date': _iso(due),
              'amount': c['rent_amount'],
              'status': ObligationStatus.unpaid,
            },
            conflictAlgorithm: ConflictAlgorithm.ignore,
          );
          if (id > 0) created++;

          cursor = DateTime(cursor.year, cursor.month + 1);
        }

        // TODO: add fixed-monthly and consumption service obligations here
        // (type = ObligationType.service(serviceId)).
      }

      // Expire contracts whose end date has passed (after generating).
      await txn.update(
        'contracts',
        {'status': ContractStatus.expired},
        where: 'status = ? AND end_date < ?',
        whereArgs: [ContractStatus.active, _iso(now!)],
      );

      // Flag late obligations.
      await txn.rawUpdate(
        'UPDATE obligations SET status = ? '
        'WHERE status IN (?, ?) AND due_date < ?',
        [
          ObligationStatus.overdue,
          ObligationStatus.unpaid,
          ObligationStatus.partial,
          _iso(now),
        ],
      );
    });

    return created;
  }
}

class DashboardStats {
  int properties = 0,
      units = 0,
      rented = 0,
      available = 0,
      tenants = 0,
      activeContracts = 0;
  double dueThisMonth = 0, paid = 0, remaining = 0, overdue = 0;
}

class HomeController extends GetxController with StateMixin<DashboardStats> {
  DatabaseService get _db => Get.find<DatabaseService>();

  @override
  void onInit() {
    super.onInit();
    load();
  }

  Future<int> _count(String sql, [List<Object?>? args]) async {
    final r = await _db.db.rawQuery(sql, args);
    return (r.first.values.first as int?) ?? 0;
  }

  Future<double> _sum(String sql, [List<Object?>? args]) async {
    final r = await _db.db.rawQuery(sql, args);
    return ((r.first.values.first as num?) ?? 0).toDouble();
  }

  Future<void> load() async {
    change(null, status: RxStatus.loading());
    try {
      final s = DashboardStats();
      final period = ObligationService.periodOf(DateTime.now());

      s.properties = await _count('SELECT COUNT(*) FROM properties');
      s.units = await _count('SELECT COUNT(*) FROM units');
      s.rented = await _count("SELECT COUNT(*) FROM units WHERE status='rented'");
      s.available =
          await _count("SELECT COUNT(*) FROM units WHERE status='available'");
      s.tenants = await _count('SELECT COUNT(*) FROM tenants');
      s.activeContracts =
          await _count("SELECT COUNT(*) FROM contracts WHERE status='active'");

      s.dueThisMonth = await _sum(
          "SELECT SUM(amount) FROM obligations WHERE period=? AND status!='cancelled'",
          [period]);
      s.paid = await _sum(
          "SELECT SUM(paid_amount) FROM obligations WHERE period=? AND status!='cancelled'",
          [period]);
      s.remaining = s.dueThisMonth - s.paid;
      s.overdue = await _sum(
          "SELECT SUM(amount - paid_amount) FROM obligations WHERE status='overdue'");

      change(s, status: RxStatus.success());
    } catch (e) {
      change(null, status: RxStatus.error(e.toString()));
    }
  }
}

class PropertiesController extends GetxController

  void onInit() {
    super.onInit();
    load();
  }

  Future<void> load() async {
    change(null, status: RxStatus.loading());
    try {
      final list = await _repo.getAll();
      change(list,
          status: list.isEmpty ? RxStatus.empty() : RxStatus.success());
    } catch (e) {
      change(null, status: RxStatus.error(e.toString()));
    }
  }

  Future<void> save(PropertyModel p) async {
    if (p.id == null) {
      await _repo.insert(p);
    } else {
      await _repo.update(p);
    }
    await load();
  }

  Future<void> remove(PropertyModel p) async {
    // Business rule: never orphan units / contracts / financial history.
    if (await _repo.unitCount(p.id!) > 0) {
      Get.snackbar('delete'.tr, 'cannot_delete_has_units'.tr);
      return;
    }
    await _repo.delete(p.id!);
    await load();
  }
}

import 'package:flutter/services.dart' show rootBundle;

class ContractPdfData {

    required this.ownerPhone,
    required this.tenantName,
    required this.tenantId,
    required this.tenantPhone,
    required this.propertyName,

    required this.rent,
    required this.currency,
    required this.deposit,
    required this.startDate,
    required this.endDate,
    required this.paymentCycle,

  });

  final String ownerName, ownerId, ownerPhone;
  final String tenantName, tenantId, tenantPhone;
  final String propertyName, propertyAddress, unitNumber;
  final double rent, deposit;
  final String currency, startDate, endDate, paymentCycle, termsText;

/// Builds a printable contract PDF. Signature, thumbprint and witness areas
/// are intentionally left blank for manual completion after printing.
class ContractPdfService {
  Future<pw.Font?> _font(String asset) async {
    try {
      return pw.Font.ttf(await rootBundle.load(asset));
    } catch (_) {
      return null; // font file missing -> Arabic will not render
    }
  }

  Future<Uint8List> build(ContractPdfData d, {bool rtl = true}) async {
    final regular = await _font('assets/fonts/Cairo-Regular.ttf');
    final bold = await _font('assets/fonts/Cairo-Bold.ttf');
    final doc = pw.Document(
      theme: (regular != null && bold != null)
          ? pw.ThemeData.withFont(base: regular, bold: bold)
          : null,
    );

    pw.Widget row(String k, String v) => pw.Padding(
          padding: const pw.EdgeInsets.symmetric(vertical: 2),
          child: pw.Row(children: [
            pw.SizedBox(
                width: 120,
                child: pw.Text(k,
                    style: pw.TextStyle(fontWeight: pw.FontWeight.bold))),
            pw.Expanded(child: pw.Text(v)),
          ]),
        );

    pw.Widget box(String label) => pw.Container(
          width: 160,
          height: 70,
          decoration: pw.BoxDecoration(border: pw.Border.all()),
          padding: const pw.EdgeInsets.all(4),
          child: pw.Text(label, style: const pw.TextStyle(fontSize: 9)),
        );

    pw.Widget sigRow(String who) => pw.Padding(
          padding: const pw.EdgeInsets.only(top: 10),
          child: pw.Column(crossAxisAlignment: pw.CrossAxisAlignment.start, children: [
            pw.Text(who, style: pw.TextStyle(fontWeight: pw.FontWeight.bold)),
            pw.SizedBox(height: 4),
            pw.Row(
              mainAxisAlignment: pw.MainAxisAlignment.spaceBetween,
              children: [box('Signature / التوقيع'), box('Thumbprint / البصمة')],
            ),
          ]),
        );

    doc.addPage(
      pw.MultiPage(
        pageFormat: PdfPageFormat.a4,
        textDirection: rtl ? pw.TextDirection.rtl : pw.TextDirection.ltr,
        margin: const pw.EdgeInsets.all(32),
        build: (_) => [
          pw.Center(
            child: pw.Text('عقد إيجار / Lease Agreement',
                style: pw.TextStyle(fontSize: 20, fontWeight: pw.FontWeight.bold)),
          ),
          pw.SizedBox(height: 16),
          pw.Text('Owner / المالك', style: pw.TextStyle(fontWeight: pw.FontWeight.bold)),
          row('Name', d.ownerName),
          row('ID', d.ownerId),
          row('Phone', d.ownerPhone),
          pw.SizedBox(height: 8),
          pw.Text('Tenant / المستأجر', style: pw.TextStyle(fontWeight: pw.FontWeight.bold)),
          row('Name', d.tenantName),
          row('ID', d.tenantId),
          row('Phone', d.tenantPhone),
          pw.SizedBox(height: 8),
          pw.Text('Property / العقار', style: pw.TextStyle(fontWeight: pw.FontWeight.bold)),
          row('Property', d.propertyName),
          row('Address', d.propertyAddress),
          row('Unit', d.unitNumber),
          pw.SizedBox(height: 8),
          pw.Text('Financial & Term / المالية والمدة',
              style: pw.TextStyle(fontWeight: pw.FontWeight.bold)),
          row('Rent', '${d.rent.toStringAsFixed(2)} ${d.currency}'),
          row('Deposit', '${d.deposit.toStringAsFixed(2)} ${d.currency}'),
          row('Cycle', d.paymentCycle),
          row('Start', d.startDate),
          row('End', d.endDate),
          pw.SizedBox(height: 8),
          pw.Text('Terms / الشروط', style: pw.TextStyle(fontWeight: pw.FontWeight.bold)),
          pw.Text(d.termsText),
          pw.SizedBox(height: 16),
          sigRow('Owner / المالك'),
          sigRow('Tenant / المستأجر'),
          sigRow('Witness 1 / الشاهد الأول:  ____________________'),
          sigRow('Witness 2 / الشاهد الثاني:  ____________________'),
          pw.SizedBox(height: 12),
          pw.Text('Date / التاريخ: ____ / ____ / ________'),
        ],
      ),
    );
    return doc.save();
  }

  Future<File> saveTo(File target, ContractPdfData d) async {
    return target.writeAsBytes(await build(d));
  }
}

# PTMS - Property & Tenant Management System

Flutter / Android / GetX / SQLite (local-first, single owner).
